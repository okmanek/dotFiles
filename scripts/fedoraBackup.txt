#!/usr/bin/bash

# ~/.config/kglobalshortcutsrc				KDE keyboard shortcuts
# ~/.config/plasma-org.kde.plasma.desktop-appletsrc	panels, widgets, desktop layouts
# ~/.config/kwinrc						window behaviour, KWin rules
# ~/.config/kdeglobals					monitor configuration
# ~/.local/share/kscreen/

paths=(.config)
for d in .local/share/kscreen .local/share/plasma .local/share/kwin .local/share/kxmlgui6; do
  [[ -e "$HOME/$d" ]] && paths+=("$d")
done
tar -czf "$HOME/fedora-kde-settings-$(date +%F).tar.gz" -C "$HOME" "${paths[@]}"
